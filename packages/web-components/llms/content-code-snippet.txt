# Content/Code Snippet

## Examples


### Code Snippet

```tsx
({
  codeSnippet,
  variant,
  height,
  context,
  sx
}) => {
  return `
    <cbp-code-snippet
      ${variant ? `variant="${variant}"` : ''}
      ${height ? `height="${height}"` : ''}
      ${context && context != 'light-inverts' ? `context="${context}"` : ''}
      ${sx ? `sx='${JSON.stringify(sx)}'` : ''}
    >
      ${codeSnippet}
    </cbp-code-snippet>
  `;
}
```
